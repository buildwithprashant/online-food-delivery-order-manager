📐 Algorithm
AssignDelivery(maxDist):
  candidates = orders
    .filter(o => !o.isPaid)
    .filter(o => o.distance ≤ maxDist)
  if candidates.length === 0:
    return "No order available"
  return min(candidates, o => o.distance)
